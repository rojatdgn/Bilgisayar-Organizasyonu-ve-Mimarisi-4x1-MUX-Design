module four_bit_adder_tb;

    reg [3:0] A;
    reg [3:0] B;
    reg Cin;
    wire [3:0] Sum;
    wire Cout;

    four_bit_adder uut (
        .A(A), 
        .B(B), 
        .Cin(Cin), 
        .Sum(Sum), 
        .Cout(Cout)
    );

    initial begin
        $monitor("Zaman = %0t | A = %b (%d) | B = %b (%d) | Cin = %b | Toplam = %b (%d) | Cout = %b", 
                 $time, A, A, B, B, Cin, Sum, Sum, Cout);

        A = 4'b0000; B = 4'b0000; Cin = 0; #10;
        
        A = 4'b0101; B = 4'b0011; Cin = 0; #10;
        
        A = 4'b1111; B = 4'b0001; Cin = 0; #10;
        
        A = 4'b0111; B = 4'b0111; Cin = 1; #10;

        $display("Test tamamlandı.");
        $finish;
    end

endmodule
